document.addEventListener("DOMContentLoaded", function () {
    const seatContainer = document.getElementById("seat-container");
    const personCountInput = document.getElementById("person-count");
    const reserveButton = document.getElementById("reserve-button");

    let selectedSeats = []; // 선택된 좌석 추적
    let maxSelectable = 0; // 선택 가능한 최대 좌석 수
    let reservedSeats = []; // 서버에서 받아오는 예약된 좌석 데이터

    // 서버에서 예약된 좌석 데이터를 가져오기
    async function fetchReservedSeats(movieId, showTime, theaterId) {
        try {
            const url = `http://localhost:8080/seats?movieId=${movieId}&showTime=${encodeURIComponent(
                showTime
            )}&theaterId=${theaterId}`;

            const response = await fetch(url);
            if (!response.ok) {
                throw new Error(`서버로부터 데이터를 가져오는 데 실패했습니다: ${response.status}`);
            }

            reservedSeats = await response.json(); // JSON 데이터를 파싱
            console.log("예약된 좌석 데이터:", reservedSeats);
            createSeats(theaterId); // 좌석 생성
        } catch (error) {
            console.error("좌석 데이터를 가져오는 중 오류 발생:", error);
        }
    }

    // 좌석 배열 생성
    function createSeats(theaterId) {
        seatContainer.innerHTML = ""; // 기존 좌석 초기화
        let rows = [];
        let columns = [];
        let colCount = 0;

        // 상영관에 따른 좌석 배열 정의
        switch (theaterId) {
            case "1": // 1관: A, B 열, 10x3 배열
                rows = ["A", "B"];
                columns = ["a", "b", "c"];
                colCount = 10; // 좌석 번호 개수
                break;
            case "2": // 2관: A, B, C 열, 10x2 배열
                rows = ["A", "B", "C"];
                columns = ["a", "b"];
                colCount = 10; // 좌석 번호 개수
                break;
            case "3": // 3관: A, B 열, 8x3 배열
                rows = ["A", "B"];
                columns = ["a", "b", "c"];
                colCount = 8; // 좌석 번호 개수
                break;
            default:
                alert("잘못된 상영관 번호입니다.");
                return;
        }

        // 좌석 UI 생성
        rows.forEach((row) => {
            columns.forEach((column) => {
                const rowDiv = document.createElement("div"); // 행 컨테이너
                rowDiv.classList.add("seat-row");
                rowDiv.textContent = `${row}열 ${column}: `; // A열 a, B열 b, ...

                for (let i = 1; i <= colCount; i++) {
                    const seatButton = document.createElement("button");
                    seatButton.textContent = i; // 열 번호(숫자)
                    seatButton.classList.add("seat");
                    seatButton.dataset.row = row; // A, B, ...
                    seatButton.dataset.column = column; // a, b, c, ...
                    seatButton.dataset.number = i; // 좌석 번호

                    // 예약된 좌석 표시
                    if (
                        reservedSeats.some(
                            (s) =>
                                s.row === row &&
                                s.column === column &&
                                s.number == i
                        )
                    ) {
                        seatButton.classList.add("reserved");
                        seatButton.disabled = true; // 예약된 좌석은 선택 불가
                    } else {
                        // 좌석 클릭 이벤트
                        seatButton.addEventListener("click", () => toggleSeatSelection(seatButton));
                    }

                    rowDiv.appendChild(seatButton);
                }

                seatContainer.appendChild(rowDiv);
            });
        });
    }

    // 좌석 선택 처리
    function toggleSeatSelection(seatButton) {
        const seat = {
            row: seatButton.dataset.row, // 행 (A, B, ...)
            column: seatButton.dataset.column, // 열 (a, b, c, ...)
            number: seatButton.dataset.number, // 좌석 번호 (1, 2, 3, ...)
        };

        if (seatButton.classList.contains("selected")) {
            // 선택 해제
            seatButton.classList.remove("selected");
            selectedSeats = selectedSeats.filter(
                (s) =>
                    !(s.row === seat.row && s.column === seat.column && s.number === seat.number)
            );
        } else {
            if (selectedSeats.length >= maxSelectable) {
                alert(`최대 ${maxSelectable}개의 좌석만 선택 가능합니다.`);
                return;
            }

            // 선택 추가
            seatButton.classList.add("selected");
            selectedSeats.push(seat);
        }

        console.log("선택된 좌석:", selectedSeats);
    }

    // 인원 수 변경 이벤트
    personCountInput.addEventListener("change", (event) => {
        const count = parseInt(event.target.value, 10);
        if (isNaN(count) || count <= 0) {
            alert("올바른 인원 수를 입력하세요.");
            return;
        }

        maxSelectable = count; // 선택 가능한 최대 좌석 수 설정
        selectedSeats = []; // 선택 초기화
        document.querySelectorAll(".seat.selected").forEach((seat) => {
            seat.classList.remove("selected");
        });
    });

    // 예약 버튼 클릭 이벤트
    reserveButton.addEventListener("click", () => {
        if (selectedSeats.length !== maxSelectable) {
            alert(`좌석 ${maxSelectable}개를 선택해주세요.`);
            return;
        }

        alert(
            `선택된 좌석: ${selectedSeats
                .map((s) => `${s.row}열 ${s.column}${s.number}`)
                .join(", ")}`
        );
        // 예약 API 호출 가능
    });

});
